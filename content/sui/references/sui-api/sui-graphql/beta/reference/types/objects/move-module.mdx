Modules are a unit of code organization in Move.

Modules belong to packages, and contain type and function definitions.


```graphql
type MoveModule {
  bytes: Base64
  datatype(
    name: String!
  ): MoveDatatype
  datatypes(
    first: Int
    after: String
    last: Int
    before: String
  ): MoveDatatypeConnection
  disassembly: String
  enum(
    name: String!
  ): MoveEnum
  enums(
    first: Int
    after: String
    last: Int
    before: String
  ): MoveEnumConnection
  fileFormatVersion: Int
  friends(
    first: Int
    after: String
    last: Int
    before: String
  ): MoveModuleConnection
  fullyQualifiedName: String!
  function(
    name: String!
  ): MoveFunction
  functions(
    first: Int
    after: String
    last: Int
    before: String
  ): MoveFunctionConnection
  name: String!
  package: MovePackage
  struct(
    name: String!
  ): MoveStruct
  structs(
    first: Int
    after: String
    last: Int
    before: String
  ): MoveStructConnection
}
```
